(function(){
  "use strict";
  var $ = function(s, c){ return (c||document).querySelector(s); };
  var $$ = function(s, c){ return Array.prototype.slice.call((c||document).querySelectorAll(s)); };

  /* ---------- Project-type tabs ---------- */
  var tabs = $$(".tab");
  function showTab(name){
    tabs.forEach(function(t){ t.setAttribute("aria-selected", t.dataset.tab === name ? "true" : "false"); });
    $$(".draw-panel").forEach(function(p){ p.classList.toggle("active", p.id === "draw-" + name); });
  }
  tabs.forEach(function(t){ t.addEventListener("click", function(){ showTab(t.dataset.tab); }); });

  /* ---------- Quote-type buttons ---------- */
  $$("[data-quote-type]").forEach(function(b){
    b.addEventListener("click", function(){
      var sel = $("#q-type");
      sel.value = b.dataset.quoteType;
      document.getElementById("quote-form").scrollIntoView({behavior:"smooth"});
      setStatus("Project type set to “" + b.dataset.quoteType + "”. Fill in the rest below.", "ok");
    });
  });


  /* ---------- Quote form ---------- */
  var photos = []; // {name, size, file} — kept in the browser only, never uploaded
  var statusEl = $("#q-status");
  function setStatus(msg, kind){
    statusEl.textContent = msg;
    statusEl.className = "form-status " + (kind || "");
  }
  function renderThumbs(){
    var box = $("#q-thumbs"); box.innerHTML = "";
    photos.forEach(function(p, i){
      var fig = document.createElement("figure");
      var img = document.createElement("img");
      img.alt = p.name;
      var reader = new FileReader();
      reader.onload = function(ev){ img.src = ev.target.result; };
      reader.readAsDataURL(p.file);
      var cap = document.createElement("figcaption");
      cap.textContent = p.name + " (" + Math.round(p.size/1024) + " KB)";
      var rm = document.createElement("button");
      rm.type = "button"; rm.textContent = "Remove";
      rm.addEventListener("click", function(){ photos.splice(i, 1); renderThumbs(); });
      fig.appendChild(img); fig.appendChild(cap); fig.appendChild(rm);
      box.appendChild(fig);
    });
  }
  function addPhotos(fileList){
    var added = 0;
    Array.prototype.forEach.call(fileList, function(f){
      if (!f.type || f.type.indexOf("image/") !== 0) return;
      if (photos.length >= 8){ setStatus("Up to 8 photos, please — remove one to add another.", "warn"); return; }
      photos.push({ name: f.name, size: f.size, file: f }); added++;
    });
    if (added) renderThumbs();
  }
  $("#q-photos").addEventListener("change", function(e){ addPhotos(e.target.files); e.target.value = ""; });
  var drop = $("#q-drop");
  ["dragover","dragenter"].forEach(function(ev){ drop.addEventListener(ev, function(e){ e.preventDefault(); drop.classList.add("over"); }); });
  ["dragleave","drop"].forEach(function(ev){ drop.addEventListener(ev, function(e){ e.preventDefault(); drop.classList.remove("over"); }); });
  drop.addEventListener("drop", function(e){ if (e.dataTransfer && e.dataTransfer.files) addPhotos(e.dataTransfer.files); });

  function collectQuote(){
    return {
      name: $("#q-name").value.trim(),
      email: $("#q-email").value.trim(),
      phone: $("#q-phone").value.trim(),
      type: $("#q-type").value,
      size: $("#q-size").value.trim(),
      location: $("#q-location").value.trim(),
      timeline: $("#q-timeline").value,
      budget: $("#q-budget").value,
      desc: $("#q-desc").value.trim(),
      photos: photos.map(function(p){ return p.name + " (" + Math.round(p.size/1024) + " KB)"; })
    };
  }
  function quoteText(q){
    var lines = [
      "Quote request — Ray's Ironworks (from rays-ironworks website)",
      "",
      "Name: " + (q.name || "(not given)"),
      "Email: " + (q.email || "(not given)"),
      "Phone: " + (q.phone || "(not given)"),
      "Project type: " + (q.type || "(not chosen)"),
      "Approximate size: " + (q.size || "(not given)"),
      "Location: " + (q.location || "(not given)"),
      "Timeline: " + q.timeline,
      "Budget range: " + q.budget,
      "",
      "Description:",
      q.desc || "(not given)",
      "",
      "Photos attached to this email by the visitor (" + q.photos.length + "):"
    ];
    q.photos.forEach(function(n){ lines.push(" - " + n); });
    if (!q.photos.length) lines.push(" (none — no photos were attached)");
    lines.push("", "— Sent from the Ray's Ironworks website quote form. Photos are attached to this email by the visitor; the website does not store them.");
    return lines.join("\n");
  }
  function validQuote(q){
    if (!q.name){ setStatus("Please add your name first.", "warn"); $("#q-name").focus(); return false; }
    if (!q.email || q.email.indexOf("@") < 1){ setStatus("Please add a valid email so Ray can reply.", "warn"); $("#q-email").focus(); return false; }
    if (!q.type){ setStatus("Please choose a project type.", "warn"); $("#q-type").focus(); return false; }
    if (!q.desc){ setStatus("Please describe your idea — even a sentence or two helps.", "warn"); $("#q-desc").focus(); return false; }
    return true;
  }
  $("#q-email-btn").addEventListener("click", function(){
    var q = collectQuote();
    if (!validQuote(q)) return;
    var subject = "Quote request — " + q.type + " — " + q.name;
    var body = quoteText(q);
    var url = "mailto:raysironworks101@gmail.com?subject=" + encodeURIComponent(subject) + "&body=" + encodeURIComponent(body);
    if (url.length > 1800){
      setStatus("Your request is long for an email link — use “Download Quote Summary” instead, then send that file to raysironworks101@gmail.com.", "warn");
      return;
    }
    window.location.href = url;
    setStatus(q.photos.length
      ? "Your email app should have opened with the request filled in. Remember to attach your " + q.photos.length + " photo(s) to the email before sending."
      : "Your email app should have opened with the request filled in. Send it whenever you are ready.", "ok");
  });
  $("#q-download-btn").addEventListener("click", function(){
    var q = collectQuote();
    if (!validQuote(q)) return;
    var blob = new Blob([quoteText(q)], { type: "text/plain" });
    var a = document.createElement("a");
    a.href = URL.createObjectURL(blob);
    a.download = "rays-ironworks-quote-request.txt";
    document.body.appendChild(a); a.click(); a.remove();
    setStatus("Quote summary downloaded. Send it (and your photos) to raysironworks101@gmail.com.", "ok");
  });
  $("#q-copy-btn").addEventListener("click", function(){
    var q = collectQuote();
    if (!validQuote(q)) return;
    var text = quoteText(q);
    function done(){ setStatus("Quote details copied. Paste them into an email to raysironworks101@gmail.com — and attach your photos.", "ok"); }
    if (navigator.clipboard && navigator.clipboard.writeText){
      navigator.clipboard.writeText(text).then(done, function(){
        var ta = document.createElement("textarea"); ta.value = text; document.body.appendChild(ta);
        ta.select(); try { document.execCommand("copy"); done(); } catch(e){ setStatus("Copy did not work in this browser — use “Download Quote Summary” instead.", "warn"); }
        ta.remove();
      });
    } else { setStatus("Copy is not available in this browser — use “Download Quote Summary” instead.", "warn"); }
  });
